SCENARIO 1.1 – STUDENT RECORDS (EXTRA QUERIES)
Database: college
Collection: students
===============================================

STEP 1 – Find students who scored more than 75 marks using $gt

db.students.find({ marks: { $gt: 75 } })

EXPECTED OUTPUT:
[
    { rollNo: 101, name: "Arun",    dept: "CSE", marks: 78, city: "Chennai" },
    { rollNo: 102, name: "Bhavna",  dept: "ECE", marks: 85, city: "Jaipur" },
    { rollNo: 105, name: "Suresh",  dept: "CSE", marks: 91, city: "Trichy" }
]


STEP 2 – Find students who scored less than or equal to 70 using $lte

db.students.find({ marks: { $lte: 70 } })

EXPECTED OUTPUT:
[
    { rollNo: 103, name: "Karthik", dept: "CSE", marks: 68, city: "Chennai" }
]


STEP 3 – Find students belonging to CSE or ECE using $in

db.students.find({ dept: { $in: ["CSE", "ECE"] } })

EXPECTED OUTPUT:
[
    { rollNo: 101, name: "Arun",    dept: "CSE", marks: 78, city: "Chennai" },
    { rollNo: 102, name: "Bhavna",  dept: "ECE", marks: 85, city: "Jaipur" },
    { rollNo: 103, name: "Karthik", dept: "CSE", marks: 68, city: "Chennai" },
    { rollNo: 105, name: "Suresh",  dept: "CSE", marks: 91, city: "Trichy" }
]


STEP 4 – Find students who are not from Chennai using $ne

db.students.find({ city: { $ne: "Chennai" } })

EXPECTED OUTPUT:
[
    { rollNo: 102, name: "Bhavna", dept: "ECE", marks: 85, city: "Jaipur" },
    { rollNo: 105, name: "Suresh", dept: "CSE", marks: 91, city: "Trichy" }
]


STEP 5 – Find students whose names start with S using $regex

db.students.find({ name: { $regex: "^S" } })

EXPECTED OUTPUT:
[
    { rollNo: 105, name: "Suresh", dept: "CSE", marks: 91, city: "Trichy" }
]


STEP 6 – Display only name and marks, sorted by marks in descending order

db.students.find(
    {},
    { _id: 0, name: 1, marks: 1 }
).sort({ marks: -1 })

EXPECTED OUTPUT:
[
    { name: "Suresh",  marks: 91 },
    { name: "Bhavna",  marks: 85 },
    { name: "Arun",    marks: 78 },
    { name: "Karthik", marks: 68 }
]


STEP 7 – Display the two students with the highest marks using sort() and limit()

db.students.find(
    {},
    { _id: 0, name: 1, marks: 1 }
).sort({ marks: -1 }).limit(2)

EXPECTED OUTPUT:
[
    { name: "Suresh", marks: 91 },
    { name: "Bhavna", marks: 85 }
]


STEP 8 – Count the number of CSE students

db.students.countDocuments({ dept: "CSE" })

EXPECTED OUTPUT:
3



SCENARIO 1.2 – EMPLOYEE DIRECTORY (EXTRA QUERIES)
Database: hr
Collection: employees
==================================================

STEP 1 – Find employees earning less than 50000 using $lt

db.employees.find({ salary: { $lt: 50000 } })

EXPECTED OUTPUT:
[
    { empId: 1, name: "Anita", dept: "HR",    salary: 42000, city: "Chennai" },
    { empId: 6, name: "Farah", dept: "HR",    salary: 45000, city: "Delhi" },
    { empId: 8, name: "Hema",  dept: "Sales", salary: 38000, city: "Hyderabad" }
]


STEP 2 – Find employees earning at least 60000 using $gte

db.employees.find({ salary: { $gte: 60000 } })

EXPECTED OUTPUT:
[
    { empId: 2, name: "Bala",   dept: "IT",      salary: 65000, city: "Bengaluru" },
    { empId: 5, name: "Elan",   dept: "IT",      salary: 72000, city: "Chennai" },
    { empId: 7, name: "Gopal",  dept: "Finance", salary: 61000, city: "Chennai" }
]


STEP 3 – Find employees who work in HR using equality condition

db.employees.find({ dept: "HR" })

EXPECTED OUTPUT:
[
    { empId: 1, name: "Anita", dept: "HR", salary: 42000, city: "Chennai" },
    { empId: 6, name: "Farah", dept: "HR", salary: 45000, city: "Delhi" }
]


STEP 4 – Find employees who are from Chennai and earn more than 60000

EXPECTED OUTPUT:
[
    { empId: 5, name: "Elan",  dept: "IT",      salary: 72000, city: "Chennai" },

]


STEP 5 – Find employees who are from Mumbai or Delhi using $or

]
